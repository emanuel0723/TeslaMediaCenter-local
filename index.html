// Tesla Drive Mode Video Bypass — tesla-bypass.js
// Works in Tesla's embedded Chromium browser
// ─────────────────────────────────────────────────────

(function TeslaBypass() {
  const video = document.querySelector('video');
  let audioCtx = null;
  let wakeLock = null;
  let userPaused = false;

  // ── 1. AudioContext — keeps the audio pipeline active
  function initAudio() {
    if (audioCtx) return;
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const src = audioCtx.createMediaElementSource(video);
    src.connect(audioCtx.destination);
  }
  // Requires user interaction to start
  document.addEventListener('click', initAudio, { once: true });

  // ── 2. Wake Lock — prevents the screen from sleeping
  async function requestWakeLock() {
    try {
      wakeLock = await navigator.wakeLock.request('screen');
      wakeLock.addEventListener('release', () => {
        // Reacquire the wake lock when possible
        if (document.visibilityState === 'visible') requestWakeLock();
      });
    } catch (e) { /* API not supported */ }
  }
  requestWakeLock();

  // ── 3. THE MAIN TRICK ─────────────────────────
  // Tesla fires visibilitychange → 'hidden'
  // when the car leaves the stop position (enters Drive/Reverse).
  // The browser automatically pauses any <video>.
  // We intercept it and force an immediate resume.
  document.addEventListener('visibilitychange', () => {
    if (document.visibilityState === 'hidden') {
      // Resume AudioContext if suspended
      if (audioCtx?.state === 'suspended') audioCtx.resume();
      // Force the video to resume
      video.play().catch(() => {});
    } else {
      // Reacquire wake lock when visible
      if (!wakeLock || wakeLock.released) requestWakeLock();
    }
  });

  // ── 4. Intercept accidental pauses
  // In addition to visibilitychange, Tesla may attempt to pause
  // the video through other means. This handler guarantees a resume.
  video.addEventListener('pause', () => {
    if (!userPaused) {
      setTimeout(() => video.play().catch(() => {}), 80);
    }
  });

  // Distinguish user pause vs system pause
  video.addEventListener('click', () => {
    userPaused = video.paused ? false : true;
  });

  // ── 5. PLEX — fetch directly from the server with a token
  async function plexFetch(serverUrl, token, path) {
    const url = `${serverUrl}${path}?X-Plex-Token=${token}`;
    const res = await fetch(url, {
      headers: {
        'Accept': 'application/json',
        'X-Plex-Token': token,
        'X-Plex-Product': 'Tesla Media Center',
        'X-Plex-Device': 'Tesla',
      }
    });
    return res.json();
  }

  // Get a direct stream URL for a Plex item
  async function getPlexStream(serverUrl, token, ratingKey) {
    const data = await plexFetch(serverUrl, token,
      `/library/metadata/${ratingKey}`);
    const part = data?.MediaContainer?.Metadata?.[0]?.Media?.[0]?.Part?.[0];
    if (!part?.key) throw new Error('Stream not found');
    return `${serverUrl}${part.key}?X-Plex-Token=${token}`;
  }

})();
